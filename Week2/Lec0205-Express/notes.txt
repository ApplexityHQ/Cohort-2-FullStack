    > How to install an express project ?
        > run: npm init -y
        > run: npm install express
        > create index.js or app.js
        > run: npx nodemon index.js

// setup for express:
const express = require('express');
const app = express();
const PORT = 3000;

app.get('/', (req, res) => {
  res.send('Hello World!');
});

app.listen(PORT, () => {
  console.log(`Server is running on http://localhost:${PORT}`);
});
